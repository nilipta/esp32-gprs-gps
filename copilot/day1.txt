You can make the ESP32 scan common modem baud rates automatically:
HardwareSerial modem(1);
int bauds[] = {
  115200,
  57600,
  38400,
  19200,
  9600
};
void setup() {
  Serial.begin(115200);
  delay(2000);
  for (int i = 0; i < sizeof(bauds)/sizeof(bauds[0]); i++) {
    Serial.printf("\nTrying %d...\n", bauds[i]);
    modem.end();
    modem.begin(bauds[i], SERIAL_8N1, 16, 17);
    modem.println("AT");
    delay(500);
    while(modem.available()) {
      Serial.write(modem.read());
    }
  }
}
void loop() {}
